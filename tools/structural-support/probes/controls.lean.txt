-- Strict transitive audit of every declaration introduced by this probe,
-- including all named controls and elaborator-generated helpers.
run_cmd do
  let env ← Lean.getEnv
  let mut count : Nat := 0
  for (name, _) in env.constants.toList do
    if (env.getModuleIdxFor? name).isNone then
      for ax in (← Lean.collectAxioms name) do
        unless [``propext, ``Classical.choice, ``Quot.sound].contains ax do
          throwError "probe declaration {name} uses unexpected axiom {ax}"
      count := count + 1
  Lean.logInfo m!"probe: {count} strict transitive checks passed"
