def branches : JointSourceExpression.Expr → Nat
  | .zero => 0
  | .atom _ => 1
  | .sum l r => branches l + branches r
  | .tensor l r => branches l * branches r

run_cmd do
  let env ← Lean.getEnv
  let mut count : Nat := 0
  for (name, _) in env.constants.toList do
    if (env.getModuleIdxFor? name).isNone then
      for ax in (← Lean.collectAxioms name) do
        unless [``propext, ``Classical.choice, ``Quot.sound].contains ax do
          throwError "additional control {name} uses unexpected axiom {ax}"
      count := count+1
  Lean.logInfo m!"additional controls: {count} strict transitive checks passed"
