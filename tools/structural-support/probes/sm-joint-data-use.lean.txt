def zeroSlot : CompactGenerationOccupations.Index 0 := ⟨0,0⟩
def balancedQ18 : SMJointSupportData.Index (fun s => if s=0 then 18 else 0) :=
  (⟨(6 : Fin 19),(6 : Fin 13)⟩,
    (zeroSlot,(zeroSlot,(zeroSlot,(zeroSlot,(zeroSlot,(zeroSlot,(zeroSlot,(zeroSlot,zeroSlot)))))))))

def lastSlot (n : Nat) : CompactGenerationOccupations.Index n :=
  ⟨⟨0,Nat.zero_lt_succ n⟩,⟨0,Nat.zero_lt_succ (n-0)⟩⟩
def lastChoice : (k : Nat) → (d : Fin (k+1) → Nat) → JointContentData.Index k d
  | 0,d => lastSlot (d 0)
  | k+1,d => (lastSlot (d 0), lastChoice k (fun i => d i.succ))
