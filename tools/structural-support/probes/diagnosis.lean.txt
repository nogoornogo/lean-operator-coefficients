instance ownerEq : (w : ColorTensorTree.Shape) → DecidableEq (ColorTensorTree.Leaf w)
  | .leaf _ _ => inferInstanceAs (DecidableEq Unit)
  | .tensor l r => by
    letI := ownerEq l
    letI := ownerEq r
    exact inferInstanceAs (DecidableEq (ColorTensorTree.Leaf l ⊕ ColorTensorTree.Leaf r))
instance positiveDecision (w : ColorTensorTree.Shape) (k : Nat)

    (s : Equiv.Perm (Fin (ColorTensorTree.negative w + 3*k))) :
    Decidable (ColorTreeDiagrams.PositiveValid w k h s) := by

def expressionSize : JointSourceExpression.Expr → Nat
  | .zero => 0
  | .atom _ => 1
  | .sum l r => expressionSize l + expressionSize r
  | .tensor l r => expressionSize l * expressionSize r

def q6 : JointSourceExpression.Expr := by
  run_tac do
    let e ← Lean.Elab.Term.elabTerm (← `( (SMSignedCoreJointPresentation.presentation 0 6).expression )) none
    let v ← Lean.Meta.reduce e
    Lean.Elab.Tactic.closeMainGoal `metadata v

theorem metadataCorrespondence :
    q6 = (SMSignedCoreJointPresentation.presentation 0 6).expression := rfl

def occupiedQ18 : JointSourceExpression.Expr :=
  .tensor (.tensor q6 q6) (.tensor q6 (.atom ⟨.symmetric 0,6,0,0⟩))

-- Diagnostic projection only: this is deliberately not equivalent to the
-- original joint support predicate. It isolates weak/left/right decisions.
def eraseColor : JointSourceExpression.Expr → JointSourceExpression.Expr
  | .zero => .zero
  | .atom a => .atom ⟨.symmetric 0,a.weak,a.leftSpin,a.rightSpin⟩
  | .sum l r => .sum (eraseColor l) (eraseColor r)
  | .tensor l r => .tensor (eraseColor l) (eraseColor r)
